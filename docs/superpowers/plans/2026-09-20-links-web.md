# `gg://` links in `gg web` (plan 3c) Implementation Plan

> **For the executing session:** this plan is executed by the ONE session that
> wrote it, task by task, in the worktree — never by subagents (CLAUDE.md).
> Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** the web gets the "compare with link…" dialog with the base picker, a link-comparison view with per-member byte sources, save comparison, and a Previews tab that lists commit pairs and comparisons beside merge previews.

**Architecture:** three thin read routes (`/api/compare-links`, `/api/link-base`, `/api/saved-compares`) and three write-guarded ones call what plan 3b-2 shipped in `domain`. The client gets one new module (`static/linkcompare.js`), one new view opener in `files.js`, and new row kinds in `previews.js`. No JS link parser, no JS rewrite: the server classifies and rewrites (W1).

**Tech stack:** Go `net/http` handlers in `internal/web`, vanilla ES modules in `internal/web/static`, real-git handler tests, source-assertion JS gates, a playwright/CDP browser probe.

**Spec:** `docs/superpowers/specs/2026-09-20-links-web-design.md` (W1–W10, F1–F10). Parents: `2026-09-19-links-tui-design.md` §3.5–3.8, `2026-09-16-unified-links-design.md` §5.1/§5.5.

## Global constraints

- **One door.** `internal/web` never calls `EvalLink` (archtest `TestFrontendsReachLinkSetsThroughTheDoor`). Link text becomes a file set only through `domain.CompareLinks`.
- **Target first.** Base `B` for ref `A` → `@B...A`. Gates assert the link TEXT.
- **Every link is `model.Link{…}.String()`**, never concatenation. A pair half is a FULL sha.
- **Words.** "pair" = a saved commit pair (a SET entry `@A..B`). A two-link entry is a **comparison**. Never mix them — in code, labels, help or docs.
- **Web rules.** An element hidden by class needs its own `#id.hidden { display: none; }` rule. Every POST/DELETE route is `writeGuard`ed. Repeatable fetches go through `runOnce`. No browser storage (W10). A copy row's label STARTS with `copy gg link` and its `act` calls `copyLink(` (F9).
- **`ResolveOpts`.** Every call passes `domain.ResolveOpts{Cwd: svc, RegistryPath: s.reposStatePath()}`. Tests set `srv.reposPath` to a file under `t.TempDir()` — never the user's registry.
- **Fixtures.** Every link fixture set has a LOCAL-form row (`gg:///abs/…`). Two look-alike arms must disagree on one fixture. A comparison only READS a member present on both sides — a byte-source test needs an `M` row.
- **Break tables.** Each task ends with one. "Watch it fail" = with the GUARD removed. A break that fails to BUILD is not red — redo it. Assert BEFORE any step that could reset state.
- **Exit codes.** `… > log 2>&1; echo EXIT=$?`. Never `go test | tail && git commit`.
- **Help** must not advertise a row before the task that adds it.

// linkRepo: main has a.txt + b.txt; feat/x edits both. Returns the dir and the
// two tips.
func linkRepo(t *testing.T) (dir, mainSha, featSha string) {
	t.Helper()
	isolateState(t)
	dir = newRepoDir(t, 1)
	for _, f := range []string{"a.txt", "b.txt"} {
		os.WriteFile(filepath.Join(dir, f), []byte("one\n"), 0o644)
	}
	gitRun(t, dir, "add", "-A")
	gitRun(t, dir, "commit", "-m", "base")
	mainSha = gitRun(t, dir, "rev-parse", "HEAD")
	gitRun(t, dir, "checkout", "-q", "-b", "feat/x")
	for _, f := range []string{"a.txt", "b.txt"} {
		os.WriteFile(filepath.Join(dir, f), []byte("two\n"), 0o644)
	}
	gitRun(t, dir, "commit", "-am", "edit both")
	featSha = gitRun(t, dir, "rev-parse", "HEAD")
	gitRun(t, dir, "checkout", "-q", "main")
	return dir, mainSha, featSha
}

func linkServe(t *testing.T, dir string) *httptest.Server {
	t.Helper()
	svc := domain.Open(dir)
	svc.UsePreviewsDir(t.TempDir())
	srv := New(svc)
	srv.reposPath = filepath.Join(t.TempDir(), "repos.toml")
	return serve(t, srv)
}

func localLink(dir, path string, tg model.LinkTarget) string {
	abs := dir
	if path != "" {
		abs = dir + "/" + path // LOCAL form: checkout and file undivided
	}
	return model.Link{Repo: model.LinkRepo{Abs: abs}, Target: tg}.String()
}

func commitTarget(sha string) model.LinkTarget {
	return model.LinkTarget{State: model.StateCommitted, Commit: sha}
}

type linkCmp struct {
	Left, Right struct{ Text, Desc, Spec string }
	Files       []struct {
		Path, Status        string
		OldPath             string `json:"old_path"`
		LeftSpec, RightSpec string
	}
	Error, Side string
}

func cmpURL(l, r string) string {
	return "/api/compare-links?" + url.Values{"left": {l}, "right": {r}}.Encode()
}

// F4 for the web: a LOCAL-form FILE link must narrow to one file. With
// EvalLink on the parsed link (no locate) it compares the whole tree: two.
func TestCompareLinksNarrowsALocalFormFileLink(t *testing.T) {
	dir, mainSha, featSha := linkRepo(t)
	ts := linkServe(t, dir)
	var got linkCmp
	code := getJSON(t, ts, cmpURL(localLink(dir, "a.txt", commitTarget(mainSha)), localLink(dir, "", commitTarget(featSha))), &got)
	if code != http.StatusOK || len(got.Files) != 1 || got.Files[0].Path != "a.txt" || got.Files[0].Status != "M" {
		t.Fatalf("code=%d files=%+v, want exactly M a.txt", code, got.Files)
	}
	if got.Left.Spec != "commit:"+mainSha || got.Right.Spec != "commit:"+featSha {
		t.Fatalf("specs = %q / %q", got.Left.Spec, got.Right.Spec)
	}
	if got.Left.Desc == "" || got.Left.Text == "" {
		t.Fatalf("a side must carry its text and description: %+v", got.Left)
	}
}

package web

import (
	"errors"

	"net/http"

	"github.com/homeend/gigagit/internal/model"

// A link comparison: two gg:// link TEXTS through domain.CompareLinks — the
// one door (spec W2). This file never evaluates a link itself.

func init() {
	RegisterRoutes(func(mux *http.ServeMux, s *Server) {
		mux.HandleFunc("GET /api/compare-links", s.handleCompareLinks)
	})
}

// linkOpts is what every link call here passes: this checkout first, then the
// repo-switcher registry — which is what makes "another checkout" reportable
// as such instead of as "unknown" (F10).
func (s *Server) linkOpts(svc *domain.Service) domain.ResolveOpts {
	return domain.ResolveOpts{Cwd: svc, RegistryPath: s.reposStatePath()}
}

// linkSideSpec spells a RESOLVED byte source in /api/entry-diff's vocabulary.
// Per KIND with no default arm (commitEntrySide's rule): a kind this wire
// cannot spell is a server bug and must not reach the browser as an empty
// commit.
func linkSideSpec(ep model.Endpoint) (string, error) {
	switch ep.Kind() {
	case model.EndpointWorkTree:
		return "worktree", nil
	case model.EndpointIndex:
		return "staged", nil
	case model.EndpointShelf:
		return "shelf:" + ep.ShelfID(), nil
	case model.EndpointCommit:
		return "commit:" + ep.Hash(), nil
	case model.EndpointPair:
		return "commit:" + ep.PairB(), nil // a change-set's members read at b (Endpoint.FileRef)
	case model.EndpointRef, model.EndpointInvalid:
		return "", fmt.Errorf("link side: endpoint kind %d has no diff spec", ep.Kind())
	default:
		return "", fmt.Errorf("link side: unknown endpoint kind %d", ep.Kind())
	}
}

// compareLinksStatus mirrors cli.compareLinkExit: grammar is the caller's
// mistake (400); any other failure that belongs to ONE side is a link that
// cannot be answered here (422); a bare failure is the comparison's own (500).
func compareLinksStatus(err error) (int, string) {
	var se *domain.LinkSideError
	if !errors.As(err, &se) {
		return http.StatusInternalServerError, ""
	}
	if errors.Is(se.Err, model.ErrLink) {
		return http.StatusBadRequest, se.Side.String()
	}
	return http.StatusUnprocessableEntity, se.Side.String()
}

type linkSideWire struct {
	Text string `json:"text"`
	Desc string `json:"desc"`
	Spec string `json:"spec"`
}

type linkFileWire struct {
	Path      string `json:"path"`
	Status    string `json:"status"`
	OldPath   string `json:"old_path,omitempty"`
	LeftSpec  string `json:"left_spec,omitempty"`
	RightSpec string `json:"right_spec,omitempty"`
}

func (s *Server) handleCompareLinks(w http.ResponseWriter, r *http.Request) {
	q := r.URL.Query()
	left, right := q.Get("left"), q.Get("right")
	if left == "" || right == "" || q.Get("id") != "" || q.Get("a") != "" || q.Get("b") != "" {
		writeErr(w, http.StatusBadRequest, errors.New("give exactly one of: left + right, id, a + b"))
		return
	}
	s.writeLinkComparison(w, r, left, right, "")
}

func (s *Server) writeLinkComparison(w http.ResponseWriter, r *http.Request, left, right, label string) {
	svc := s.service()
	ctx := r.Context()
	c, err := svc.CompareLinks(ctx, left, right, s.linkOpts(svc))
	if err != nil {
		code, side := compareLinksStatus(err)
		w.Header().Set("Content-Type", "application/json; charset=utf-8")
		w.WriteHeader(code)
		_ = json.NewEncoder(w).Encode(map[string]string{"error": err.Error(), "side": side})
		return
	}
	side := func(text string, fs domain.FileSet) (linkSideWire, error) {
		spec, err := linkSideSpec(fs.Endpoint())
		desc := text
		if l, perr := model.ParseLink(text); perr == nil {
			desc = svc.DescribeLink(ctx, l)
		}
		return linkSideWire{Text: text, Desc: desc, Spec: spec}, err
	}
	lw, err := side(c.LeftText, c.Left)
	if err != nil {
		writeErr(w, http.StatusInternalServerError, err)
		return
	}
	rw, err := side(c.RightText, c.Right)
	if err != nil {
		writeErr(w, http.StatusInternalServerError, err)
		return
	}
	files := make([]linkFileWire, len(c.Files))
	for i, f := range c.Files {
		oldP := f.Path
		if f.OldPath != "" {
			oldP = f.OldPath
		}
		row := linkFileWire{Path: f.Path, Status: f.Status, OldPath: f.OldPath}
		// The web's compareSides: a member's bytes come from Source(path), which
		// differs from the side's endpoint for a `-u` stash's untracked file.
		ls, lerr := linkSideSpec(c.Left.Source(oldP))
		rs, rerr := linkSideSpec(c.Right.Source(f.Path))
		if lerr != nil || rerr != nil {
			writeErr(w, http.StatusInternalServerError, errors.Join(lerr, rerr))
			return
		}
		if ls != lw.Spec {
			row.LeftSpec = ls
		}
		if rs != rw.Spec {
			row.RightSpec = rs
		}
		files[i] = row
	}
	out := map[string]any{"left": lw, "right": rw, "files": files}
	if label != "" {
		out["label"] = label
	}
	writeJSON(w, out)
}

// texts the server ran through domain.CompareLinks. Same screen as an entry

  state.detailGen++;

    b: body.right.desc,
    aHash: "",
    bHash: "",
    aSpec: body.left.spec,
    bSpec: body.right.spec,
    links: { left: body.left.text, right: body.right.text },
    all: body.files || [],
    filter: "all",
    originsError: "",

  state.filesMode = "compare";
  state.fileSha = null;
  enterFilesStage();
  $("files-title").textContent = (body.label ? body.label + " — " : "") + state.compare.a + " ↔ " + state.compare.b;

// handleLinkBase is the base picker's whole server half (spec W1): the
// LOCATED kind and the suggestion — and, given a base, the rewritten link.
// The rewrite is model.Link.WithBase, the ONE rewrite; the page has none.
func (s *Server) handleLinkBase(w http.ResponseWriter, r *http.Request) {
	q := r.URL.Query()
	text := strings.TrimSpace(q.Get("link"))
	if text == "" {
		writeErr(w, http.StatusBadRequest, errors.New("link required"))
		return
	}
	l, err := model.ParseLink(text)
	if err != nil {
		writeJSON(w, map[string]string{"kind": "none"}) // no row; the comparison reports the grammar
		return
	}
	svc := s.service()
	sug, err := svc.SuggestBase(r.Context(), l, s.linkOpts(svc))
	if err != nil {
		writeErr(w, http.StatusInternalServerError, err)
		return
	}
	base := strings.TrimSpace(q.Get("base"))
	if base == "" {
		writeJSON(w, map[string]string{"kind": linkBoundKindWire[sug.Kind], "base": sug.Base, "why": sug.Why})
		return
	}
	if sug.Kind == model.LinkBoundNone {
		writeErr(w, http.StatusUnprocessableEntity, errors.New("this link cannot take a base"))
		return
	}
	out, ok := l.WithBase(base, sug.Self)
	if !ok {
		writeErr(w, http.StatusUnprocessableEntity, errors.New("that base cannot bound this link"))
		return
	}
	writeJSON(w, map[string]string{"link": out.String()})
}
